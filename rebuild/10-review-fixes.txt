// 3. galPsOvCells (the small gallery inside each overview row)  >  OnSelect
//    The checker reports a HIGH "Select cycle" here. This version opens the employee/year
//    directly instead of selecting the parent gallery.
Set(psSelected, LookUp(psEmployees, ID = ThisItem.EmpId));
Set(psLocked, psSelected.Kind = "Mobility" && psSelected.AddressCode <> "Outside");
Set(psConfirmRemove, false);
ClearCollect(psStatus, Filter(psAllStatus, Employee.Id = ThisItem.EmpId));
Set(psFY, ThisItem.FY);
Set(psRecord, LookUp(psStatus, Year(Date) = psFY));
Set(psPrev, LookUp(psStatus, Year(Date) = psFY - 1));
Set(psRegionalOverride, Blank());
Set(psConfirmResign, false);
Set(psShowOverview, false)

// 4. lblBtnMenu  >  OnSelect   (OVERVIEW)
//    The checker could not send Text(emp.ID) to SharePoint inside ForAll; work out the
//    text first so the filter compares the column with a plain value.
Clear(psAllStatus);
ForAll(
    psEmployees As emp,
    With(
        {ref: Text(emp.ID)},
        Collect(psAllStatus, Filter(EmployeeStatusTable, Employee.Value = ref))
    )
);
Set(psOvPick, 0);
Set(psShowOverview, true)
