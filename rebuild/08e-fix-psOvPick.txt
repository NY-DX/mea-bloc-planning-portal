// lblBtnMenu  >  OnSelect
Clear(psAllStatus);
ForAll(
    psEmployees As emp,
    Collect(psAllStatus, Filter(EmployeeStatusTable, Employee.Id = emp.ID))
);
Set(psOvPick, 0);
Set(psShowOverview, true)

// lblPsOvName (inside galPsOverview)  >  OnSelect
Set(psOvPick, 0);
Select(Parent)

// galPsOverview  >  OnSelect
Set(psSelected, LookUp(psEmployees, ID = ThisItem.ID));
Set(psLocked, psSelected.Kind = "Mobility" && psSelected.AddressCode <> "Outside");
Set(psConfirmRemove, false);
ClearCollect(psStatus, Filter(psAllStatus, Employee.Id = ThisItem.ID));
If(psOvPick > 0, Set(psFY, psOvPick));
Set(psRecord, LookUp(psStatus, Year(Date) = psFY));
Set(psPrev, LookUp(psStatus, Year(Date) = psFY - 1));
Set(psRegionalOverride, Blank());
Set(psConfirmResign, false);
Set(psShowOverview, false)
